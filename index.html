create unique index if not exists leaderboard_bladewar_player_id_key on public.leaderboard_bladewar (player_id); create index if not exists leaderboard_bladewar_score_idx on public.leaderboard_bladewar (score desc); alter table public.leaderboard_bladewar enable row level security; drop policy if exists "leaderboard_bladewar_select_anon" on public.leaderboard_bladewar; create policy "leaderboard_bladewar_select_anon" on public.leaderboard_bladewar for select to anon, authenticated using (true); -- insert/update HANYA lewat function submit_leaderboard_bladewar di bawah (security definer), -- jadi nggak perlu policy insert/update terpisah buat anon di sini. -- submit skor: player_id yang sama cuma nimpa baris lama KALAU skor barunya lebih tinggi. -- Formula poin (harus sama persis dengan computePoints() di client): -- poin = level*50 + kills*10 + floor(waktu_bertahan_detik/3) create or replace function public.submit_leaderboard_bladewar( p_player_id text, p_name text, p_level integer, p_time integer, p_kills integer ) returns void language plpgsql security definer set search_path = public as $$ declare new_score bigint := greatest(p_level,1)::bigint * 50 + greatest(p_kills,0)::bigint * 10 + floor(least(greatest(p_time,0), 999999) / 3)::bigint; begin insert into public.leaderboard_bladewar (player_id, name, level, time_survived, kills, score) values ( p_player_id, left(coalesce(p_name,'Player'), 20), greatest(p_level,1), greatest(p_time,0), greatest(p_kills,0), new_score ) on conflict (player_id) do update set name = excluded.name, level = excluded.level, time_survived = excluded.time_survived, kills = excluded.kills, score = excluded.score, created_at = now() where excluded.score > public.leaderboard_bladewar.score; end; $$; grant execute on function public.submit_leaderboard_bladewar(text,text,integer,integer,integer) to anon; -- ============================================================================ -- 2) MATCH HISTORY (ditampilkan di halaman Profil — "Riwayat Main") -- ============================================================================ create table if not exists public.match_history_bladewar ( id bigint generated always as identity primary key, player_id text not null, mode text not null default 'survival', -- 'survival' | 'multiplayer' level integer not null default 1, time_survived integer not null default 0, kills integer not null default 0, points integer not null default 0, created_at timestamptz not null default now() ); create index if not exists match_history_bladewar_player_idx on public.match_history_bladewar (player_id, created_at desc); alter table public.match_history_bladewar enable row level security; drop policy if exists "match_history_bladewar_select_anon" on public.match_history_bladewar; create policy "match_history_bladewar_select_anon" on public.match_history_bladewar for select to anon, authenticated using (true); create or replace function public.log_match_bladewar( p_player_id text, p_name text, p_mode text, p_level integer, p_time integer, p_kills integer ) returns void language plpgsql security definer set search_path = public as $$ declare pts bigint := greatest(p_level,1)::bigint * 50 + greatest(p_kills,0)::bigint * 10 + floor(least(greatest(p_time,0), 999999) / 3)::bigint; begin insert into public.match_history_bladewar (player_id, mode, level, time_survived, kills, points) values (p_player_id, coalesce(p_mode,'survival'), greatest(p_level,1), greatest(p_time,0), greatest(p_kills,0), pts); end; $$; grant execute on function public.log_match_bladewar(text,text,text,integer,integer,integer) to anon; -- ============================================================================ -- 3) ROOMS (multiplayer) — room punya baris beneran di database: waktu mulai (shared timer), -- siapa pembuatnya, boss apa aja yang udah tumbang, statistik tiap pemain, dan dipakai buat -- nampilin "daftar room aktif" di layar Multiplayer (berapa orang main & udah menit keberapa). -- ============================================================================ create table if not exists public.rooms_bladewar ( room_code text primary key, created_at timestamptz not null default now(), created_by text, last_active_at timestamptz not null default now(), ended_at timestamptz, bosses_defeated jsonb not null default '[]'::jsonb ); create table if not exists public.room_players_bladewar ( room_code text not null references public.rooms_bladewar(room_code) on delete cascade, player_id text not null, name text not null default 'Player', kills integer not null default 0, deaths integer not null default 0, points integer not null default 0, level integer not null default 1, last_seen timestamptz not null default now(), primary key (room_code, player_id) ); -- kalau tabel ini udah ada dari versi sebelumnya (sebelum kolom level ditambahin) alter table public.room_players_bladewar add column if not exists level integer not null default 1; alter table public.rooms_bladewar enable row level security; alter table public.room_players_bladewar enable row level security; drop policy if exists "rooms_bladewar_select_anon" on public.rooms_bladewar; create policy "rooms_bladewar_select_anon" on public.rooms_bladewar for select to anon, authenticated using (true); drop policy if exists "room_players_bladewar_select_anon" on public.room_players_bladewar; create policy "room_players_bladewar_select_anon" on public.room_players_bladewar for select to anon, authenticated using (true); -- room_join_or_create_bladewar: dipanggil begitu masuk sebuah room (klik "Buat Room Baru" atau -- klik salah satu room di daftar "Room Aktif"). Room baru -> pemanggil jadi "pembuat". Room yang -- udah ada -> ikut gabung tanpa mengubah created_at, biar timer/wave/boss semua pemain nyambung -- dari waktu yang sama. Room yang udah lama kosong (15 menit tanpa heartbeat) otomatis ditutup. create or replace function public.room_join_or_create_bladewar( p_room text, p_player_id text, p_name text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.rooms_bladewar; is_new boolean := false; begin select * into r from public.rooms_bladewar where room_code = p_room for update; if not found then insert into public.rooms_bladewar (room_code, created_by) values (p_room, p_player_id) returning * into r; is_new := true; end if; if not is_new and r.ended_at is null and r.last_active_at < now() - interval '15 minutes' then update public.rooms_bladewar set ended_at = now() where room_code = p_room returning * into r; end if; if r.ended_at is null then update public.rooms_bladewar set last_active_at = now() where room_code = p_room; insert into public.room_players_bladewar (room_code, player_id, name) values (p_room, p_player_id, left(coalesce(p_name,'Player'), 20)) on conflict (room_code, player_id) do update set name = excluded.name, last_seen = now(); end if; return jsonb_build_object( 'room_code', r.room_code, 'created_at', r.created_at, 'created_by', r.created_by, 'ended_at', r.ended_at, 'bosses_defeated', r.bosses_defeated ); end; $$; grant execute on function public.room_join_or_create_bladewar(text,text,text) to anon; -- room_bump_stats_bladewar: nambahin delta kill/mati/poin (atomic) + nyimpen level pemain saat ini, -- SEKALIGUS dipanggil terus-menerus tiap beberapa detik sebagai "heartbeat" (walau delta-nya nol) — -- ini yang bikin last_seen selalu fresh, jadi daftar "Room Aktif" nunjukin jumlah pemain yang -- beneran masih main saat ini, dan level pemain tertinggi di room itu buat ditampilin di daftar. drop function if exists public.room_bump_stats_bladewar(text,text,text,integer,integer,integer); create or replace function public.room_bump_stats_bladewar( p_room text, p_player_id text, p_name text, p_dkills integer default 0, p_ddeaths integer default 0, p_dpoints integer default 0, p_level integer default 1 ) returns void language plpgsql security definer set search_path = public as $$ begin update public.rooms_bladewar set last_active_at = now() where room_code = p_room and ended_at is null; insert into public.room_players_bladewar (room_code, player_id, name, kills, deaths, points, level, last_seen) values (p_room, p_player_id, left(coalesce(p_name,'Player'), 20), greatest(p_dkills,0), greatest(p_ddeaths,0), greatest(p_dpoints,0), greatest(p_level,1), now()) on conflict (room_code, player_id) do update set name = excluded.name, kills = public.room_players_bladewar.kills + greatest(p_dkills,0), deaths = public.room_players_bladewar.deaths + greatest(p_ddeaths,0), points = public.room_players_bladewar.points + greatest(p_dpoints,0), level = greatest(p_level,1), last_seen = now(); end; $$; grant execute on function public.room_bump_stats_bladewar(text,text,text,integer,integer,integer,integer) to anon; -- room_add_boss_bladewar: dicatet sekali tiap boss (shared) tumbang, buat rekap akhir room. create or replace function public.room_add_boss_bladewar( p_room text, p_boss_name text ) returns void language plpgsql security definer set search_path = public as $$ begin update public.rooms_bladewar set bosses_defeated = bosses_defeated || jsonb_build_object('name', left(coalesce(p_boss_name,'Boss'), 40), 'at', now()), last_active_at = now() where room_code = p_room and ended_at is null; end; $$; grant execute on function public.room_add_boss_bladewar(text,text) to anon; -- room_end_bladewar: cuma pembuat room yang boleh manggil ini buat nutup room permanen. create or replace function public.room_end_bladewar( p_room text, p_player_id text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.rooms_bladewar; begin select * into r from public.rooms_bladewar where room_code = p_room for update; if not found then return jsonb_build_object('ok', false, 'reason', 'not_found'); end if; if r.ended_at is not null then return jsonb_build_object('ok', true, 'already_ended', true); end if; if r.created_by is distinct from p_player_id then return jsonb_build_object('ok', false, 'reason', 'not_creator'); end if; update public.rooms_bladewar set ended_at = now() where room_code = p_room; return jsonb_build_object('ok', true); end; $$; grant execute on function public.room_end_bladewar(text,text) to anon; -- list_active_rooms_bladewar: buat popup "Main" -> daftar "Room Aktif" (klik salah satu buat -- langsung gabung). Menunjukin udah berjalan berapa menit, berapa orang lagi main (heartbeat 25 -- detik terakhir), dan level tertinggi di antara pemain yang lagi di dalam. Sekalian nutup otomatis -- room yang kosong 15+ menit biar nggak numpuk. drop function if exists public.list_active_rooms_bladewar(); create or replace function public.list_active_rooms_bladewar() returns table(room_code text, elapsed_seconds integer, player_count integer, top_level integer) language plpgsql security definer set search_path = public as $$ begin update public.rooms_bladewar set ended_at = now() where ended_at is null and last_active_at < now() - interval '15 minutes'; return query select r.room_code, floor(extract(epoch from (now()-r.created_at)))::integer as elapsed_seconds, (select count(*)::integer from public.room_players_bladewar rp where rp.room_code = r.room_code and rp.last_seen > now() - interval '25 seconds') as player_count, (select coalesce(max(rp.level),1)::integer from public.room_players_bladewar rp where rp.room_code = r.room_code and rp.last_seen > now() - interval '25 seconds') as top_level from public.rooms_bladewar r where r.ended_at is null order by r.last_active_at desc limit 30; end; $$; grant execute on function public.list_active_rooms_bladewar() to anon; create index if not exists room_players_bladewar_points_idx on public.room_players_bladewar (room_code, points desc);
